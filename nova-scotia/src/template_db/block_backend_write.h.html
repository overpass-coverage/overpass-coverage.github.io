5890              to_write, cur_from, to_write.size()-1, source.ptr, dest.ptr, total_size, file_handler);
5900          file_handler.insert_block((uint64*)dest.ptr);
5910        }
592
5930        flush_segment(to_write.back(), source.ptr, dest.ptr, file_handler);
594
5950        to_write.clear();
5960        cur_from = 0;
5970        total_size = 0;
5980      }
5990      else if (to_write.size() - cur_from > 1
6000          && to_write.back().size + to_write[to_write.size()-2].size > file_handler.block_size - 4)
6010      {
6020        force_flush_group(
6030            to_write, cur_from, to_write.size()-1, source.ptr, dest.ptr, total_size, file_handler);
6040        file_handler.insert_block((uint64*)dest.ptr);
605
6060        to_write.erase(to_write.begin(), to_write.end()-1);
6070        cur_from = 0;
6080        total_size = to_write.back().size;
6090      }
6100      else
6110        total_size += to_write.back().size;
6120    }
613
6140    while (total_size >= (file_handler.block_size - 4)*2)
6150    {
6160      uint32 j = cur_from;
6170      uint32 partial_size = 0;
6180      while (j < to_write.size() && partial_size <= (file_handler.block_size - 4)*2/3)
6190        partial_size += to_write[j++].size;
6200      if (partial_size > file_handler.block_size - 4)
6210        partial_size -= to_write[--j].size;
622
6230      build_dest_block(to_write, cur_from, j, source.ptr, dest.ptr, file_handler);
6240      file_handler.insert_block((uint64*)dest.ptr);
6250      cur_from = j;
6260      total_size -= partial_size;
6270    }
628
6290    if (file_handler.block_size < cur_from * 8)
6300    {
6310      to_write.erase(to_write.begin(), to_write.begin() + cur_from);
6320      cur_from = 0;
6330    }
6340  }
6350  while (src_pos < source_end)
6360  {
6370    Existing_Idx_Info< Index, Object_Predicate > info(source.ptr, src_pos, del_match.seek(src_pos + 4));
6380    if (obj_count && del_match.was_equal())
6390      (*obj_count)[del_match.idx()] = info.count;
6400    if (info.count.after > 0)
6410    {
6420      to_write.push_back({ std::move(info.existing), info.size });
6430      total_size += to_write.back().size;
6440    }
6450    src_pos = source.ptr + *(uint32*)src_pos;
6460  }
647
6480  if (total_size > 0)
6490  {
6500    force_flush_group(
6510        to_write, cur_from, to_write.size(), source.ptr, dest.ptr, total_size, file_handler);
6520    file_handler.replace_block((uint64*)dest.ptr);
6530  }
6540  else
6550    file_handler.erase_block();
6560}

674template< typename File_Blocks, typename Iterator >
675bool read_block_or_blocks(
676    File_Blocks& file_blocks, Iterator& file_it, uint32 block_size,
677    Void64_Pointer< uint64 >& source, uint32& buffer_size)
6780{
6790  file_blocks.read_block(file_it, source.ptr);
680
6810  if (*(((uint32*)source.ptr) + 1) > block_size)
6820  {
6830    uint32 new_buffer_size = (*(((uint32*)source.ptr) + 1)/block_size + 1) * block_size;
6840    if (buffer_size < new_buffer_size)
6850    {
6860      Void64_Pointer< uint64 > new_source_buffer(new_buffer_size);
6870      memcpy(new_source_buffer.ptr, source.ptr, block_size);
6880      source.swap(new_source_buffer);
689
6900      buffer_size = new_buffer_size;
6910    }
6920    for (uint i_offset = block_size; i_offset < new_buffer_size; i_offset += block_size)
6930    {
6940      ++file_it;
6950      file_blocks.read_block(file_it, source.ptr + i_offset/8, false);
6960    }
697
6980    return true;
6990  }
7000  return false;
7010}

704template< typename Object_Predicate >
705Delta_Count skip_deleted_objects(
706    uint64* source_start_ptr, uint64* dest_start_ptr,
707    const Object_Predicate& objs_to_delete, uint32 idx_size)
7080{
7090  Delta_Count count;
710
7110  uint32 src_obj_offset = 8 + idx_size;
7120  uint32 dest_obj_offset = src_obj_offset;
7130  uint32 src_size = *(uint32*)source_start_ptr;
7140  memcpy(((uint8*)dest_start_ptr) + 8, ((uint8*)source_start_ptr) + 8, idx_size);
715
7160  while (src_obj_offset < src_size)
7170  {
7180    typename Object_Predicate::Base_Object obj(((uint8*)source_start_ptr) + src_obj_offset);
7190    uint32 obj_size = obj.size_of();
720
7210    if (!objs_to_delete.match(obj))
7220    {
7230      memcpy(
7240          ((uint8*)dest_start_ptr) + dest_obj_offset, ((uint8*)source_start_ptr) + src_obj_offset, obj_size);
7250      dest_obj_offset += obj_size;
7260      ++count.after;
7270    }
7280    src_obj_offset += obj_size;
7290    ++count.before;
7300  }
731
7320  *(uint32*)dest_start_ptr = dest_obj_offset;
7330  *(((uint32*)dest_start_ptr)+1) = dest_obj_offset;
7340  memcpy(dest_start_ptr + 1, source_start_ptr + 1, idx_size);
735
7360  return count;
7370}

740template< typename Iterator >
741uint64 append_insertables(
742    uint64* dest_start_ptr, uint32 block_size, Iterator& cur_insert, const Iterator& cur_end)
7430{
7440  uint32 obj_append_offset = *(uint32*)dest_start_ptr;
7450  uint64 counter = 0;
746
7470  while ((cur_insert != cur_end) && (obj_append_offset + cur_insert->size_of() < block_size))
7480  {
7490    cur_insert->to_data(((uint8*)dest_start_ptr) + obj_append_offset);
7500    obj_append_offset += cur_insert->size_of();
7510    ++cur_insert;
7520    ++counter;
7530  }
754
7550  *(uint32*)dest_start_ptr = obj_append_offset;
7560  *(((uint32*)dest_start_ptr)+1) = obj_append_offset;
757
7580  return counter;

762template< typename Index, typename Del_Container, typename Ins_Container, typename File_Blocks >
763void update_segments(
764    File_Handler< Index, File_Blocks >& file_handler,
765    const std::map< Index, Del_Container >& to_delete, const std::map< Index, Ins_Container >& to_insert,
766    std::map< Index, Delta_Count >* obj_count)
7670{
7680  file_handler.file_it.start_segments_mode();
769
7700  uint32 buffer_size = file_handler.block_size;
7710  Void64_Pointer< uint64 > source(buffer_size);
7720  Void64_Pointer< uint64 > dest(buffer_size);
773
7740  Index idx = file_handler.file_it.block().index;
7750  auto delete_it = to_delete.find(idx);
7760  auto insert_it = to_insert.find(idx);
7770  uint32 idx_size = idx.size_of();
778
7790  Delta_Count count;
780
7810  decltype(insert_it->second.begin()) cur_insert;
7820  if (insert_it != to_insert.end())
7830    cur_insert = insert_it->second.begin();
784
7850  while (!file_handler.file_it.is_end() && file_handler.file_it.block().index == idx)
7860  {
7870    auto delta_it = file_handler.file_it;
7880    bool oversized = read_block_or_blocks(
7890        file_handler.file_blocks, file_handler.file_it, file_handler.block_size, source, buffer_size);
7900    if (oversized)
7910    {
7920      ++file_handler.file_it;
7930      typename Del_Container::Base_Object obj(((uint8*)source.ptr) + 8 + idx_size);
7940      if (delete_it != to_delete.end() && delete_it->second.match(obj))
7950        file_handler.file_blocks.erase_blocks(delta_it, file_handler.file_it);
7960      else
7970        ++count.after;
7980      ++count.before;
7990    }
8000    else
8010    {
8020      if (*(uint32*)source.ptr < 8 + idx_size)
8030      { // something has seriously gone wrong - such a block shuld not exist
8040        ++file_handler.file_it;
8050        continue;
8060      }
807
8080      if (delete_it != to_delete.end())
8090        count += skip_deleted_objects(source.ptr, dest.ptr, delete_it->second, idx_size);
8100      else
8110        memcpy(dest.ptr, source.ptr, *(uint32*)source.ptr);
812
8130      if (*(uint32*)source.ptr != *(uint32*)dest.ptr
8140          || (insert_it != to_insert.end() && *(uint32*)source.ptr < file_handler.block_size/2))
8150      {
8160        if (insert_it != to_insert.end())
8170          count.after += append_insertables(dest.ptr, file_handler.block_size, cur_insert, insert_it->second.end());
8180        flush_or_delete_block(dest.ptr, *(uint32*)dest.ptr, file_handler, idx_size);
8190      }
8200      else
8210        ++file_handler.file_it;
8220    }
8230  }
824
8250  uint8* pos = ((uint8*)dest.ptr) + idx_size + 8;
8260  memcpy(dest.ptr+1, source.ptr+1, idx_size);
8270  if (insert_it != to_insert.end())
8280  {
8290    while (cur_insert != insert_it->second.end())
8300    {
8310      flush_if_necessary_and_write_obj(
8320          dest.ptr, pos, file_handler.file_blocks, file_handler.file_it, idx, *cur_insert, file_handler.block_size, file_handler.data_filename);
8330      ++cur_insert;
8340      ++count.after;
8350    }
8360  }
8370  if (pos > ((uint8*)dest.ptr) + idx_size + 8)
8380  {
8390    *(uint32*)dest.ptr = pos - (uint8*)dest.ptr;
8400    *(((uint32*)dest.ptr)+1) = pos - (uint8*)dest.ptr;
8410    file_handler.insert_block(dest.ptr);
8420  }
843
8440  if (obj_count)
8450    (*obj_count)[idx] += count;
8460  file_handler.file_it.end_segments_mode();
8470}

852void Block_Backend< Index, Object, Iterator >::update(
853    const std::map< Index, std::set< Object > >& to_delete,
854    const std::map< Index, Container >& to_insert,
855    std::map< Index, Delta_Count >* obj_count)

8600  {

8620      relevant_idxs.push_back((iit++)->first);
8630    if (iit != to_insert.end() && iit->first == dit->first)
8640      ++iit;
8650    relevant_idxs.push_back(dit->first);
8660  }
8670  while (iit != to_insert.end())
8680    relevant_idxs.push_back((iit++)->first);

8700  File_Handler< Index, File_Blocks_ > handler(file_blocks, relevant_idxs, block_size, data_filename);

8730  for (const auto& i : to_delete)
8740    to_delete_.insert(std::make_pair(i.first, Object_Set_Predicate< Object >(i.second)));
875

8770  {
8780    if (handler.file_it.block_type() == File_Block_Index_Entry< Index >::EMPTY)
