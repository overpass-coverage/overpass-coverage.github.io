165template< typename Ref_Entry >
166inline void reorganize_tag_split_level(
167    File_Blocks_Index_Base* tags_file, uint target_level, const std::string& key, const std::string& value)
168{
169  Ranges< Tag_Index_Global > ranges{ Tag_Index_Global{ key, value }, Tag_Index_Global{ key, value + (char)0 } };
170  
171  std::map< Tag_Index_Global, std::set< Ref_Entry > > obj_to_delete;
172  {
173    Block_Backend< Tag_Index_Global, Ref_Entry > db(tags_file);
174    for (auto it = db.range_begin(ranges); !(it == db.range_end()); ++it)
175      obj_to_delete[it.index()].insert(it.object());
176  }
177  
178  std::map< Tag_Index_Global, std::vector< Ref_Entry > > obj_to_insert;
179  for (auto it = obj_to_delete.begin(); it != obj_to_delete.end(); ++it)
180  {
181    for (auto it2 = it->second.begin(); it2 != it->second.end(); ++it2)
182      obj_to_insert[Tag_Index_Global_KVI(key, value, it2->idx.val() & (0xffffffff<<(32-target_level)))]
183          .push_back(*it2);
184  }
185  
186  Block_Backend< Tag_Index_Global, Ref_Entry > db(tags_file);
187  db.update(obj_to_delete, obj_to_insert);
188}

191template< typename Ref_Entry >
192struct Freq_Tags_Updater
193{
194  Freq_Tags_Updater(File_Blocks_Index_Base* tags_file_, File_Blocks_Index_Base* freq_file_)
195    : tags_file(tags_file_), freq_file(freq_file_) {}
196  
197  void reorganize_tags_if_necessary(const std::pair< Tag_Index_Global, Delta_Count >& tag_cnt)
198  {
199    uint level = calc_tag_split_level(tag_cnt.second.after);
200    if (level > 0)
201    {
202      reorganize_tag_split_level< Ref_Entry>(tags_file, level, tag_cnt.first.key, tag_cnt.first.value);
203      freq_to_insert[{ tag_cnt.first.key }].push_back({ tag_cnt.first.value, (uint64)tag_cnt.second.after, level });
204    }
205  }
206
207  void reorganize_tags_if_necessary(
208      const std::string& key, const Frequent_Value_Entry& entry, uint64 kv_cnt)
209  {
210    uint level = calc_tag_split_level(kv_cnt);
211    if (entry.level < level)
212    {
213      reorganize_tag_split_level< Ref_Entry>(tags_file, level, key, entry.value);
214      freq_to_delete[{ key }].insert({ entry.value, (uint64)entry.count, level });
215      freq_to_insert[{ key }].push_back({ entry.value, (uint64)kv_cnt, level });
216    }
217  }
218  
219  void write_to_file()
220  {
221    Block_Backend< String_Index, Frequent_Value_Entry > db(freq_file);
222    db.update(freq_to_delete, freq_to_insert);
223  }
224
225  ~Freq_Tags_Updater()
226  {
227    if (freq_file)
228      write_to_file();
229  }
230
231  File_Blocks_Index_Base* tags_file;
232  File_Blocks_Index_Base* freq_file;
233  std::map< String_Index, std::set< Frequent_Value_Entry > > freq_to_delete;
234  std::map< String_Index, std::vector< Frequent_Value_Entry > > freq_to_insert;
235};

238template< typename Ref_Entry >
239void update_global_tags(
240    File_Blocks_Index_Base* tags_file, File_Blocks_Index_Base* freq_file,
241    std::map< Tag_Index_Global, std::set< Ref_Entry > >& obj_to_delete,
242    std::map< Tag_Index_Global, std::vector< Ref_Entry > >& obj_to_insert)
243{
244  std::map< std::string, std::vector< Frequent_Value_Entry > > frequent;
245  {
246    Block_Backend< String_Index, Frequent_Value_Entry > db(freq_file);
247    for (auto it = db.flat_begin(); !(it == db.flat_end()); ++it)
248      frequent[it.index().key].push_back(it.object());
249  }
250  for (auto& i : frequent)
251    std::sort(i.second.begin(), i.second.end());
252
253  adapt_data_to_actual_idx(obj_to_delete, frequent);
254  adapt_data_to_actual_idx(obj_to_insert, frequent);
255  
256  std::map< Tag_Index_Global, Delta_Count > cnt;
257  {
258    Block_Backend< Tag_Index_Global, Ref_Entry > db(tags_file);
259    db.update(obj_to_delete, obj_to_insert, &cnt);
260  }
261
262  Freq_Tags_Updater< Ref_Entry > freq_upd{ tags_file, freq_file };
263  auto it_cnt = cnt.begin();
264  for (auto& i : frequent)
265  {
266    while (it_cnt != cnt.end() && it_cnt->first.key < i.first)
267      freq_upd.reorganize_tags_if_necessary(*it_cnt++);
268
269    if (it_cnt != cnt.end() && it_cnt->first.key == i.first)
270    {
271      for (auto& j : i.second)
272      {
273        while (it_cnt != cnt.end() && it_cnt->first.key == i.first && it_cnt->first.value < j.value)
274          freq_upd.reorganize_tags_if_necessary(*it_cnt++);
275
276        uint64 kv_cnt = j.count;
277        while (it_cnt != cnt.end() && it_cnt->first.key == i.first && it_cnt->first.value == j.value)
278        {
279          kv_cnt += it_cnt->second.after;
280          kv_cnt -= it_cnt->second.before;
281          ++it_cnt;
282        }
283        if (kv_cnt != j.count)
284          freq_upd.reorganize_tags_if_necessary(i.first, j, kv_cnt);
285      }
286    }
287  }
288  while (it_cnt != cnt.end())
289    freq_upd.reorganize_tags_if_necessary(*it_cnt++);
290  
291  freq_upd.write_to_file();
292}

321template< typename Ref_Entry, typename Target_Idx >
322struct Migrate_Loop_Flush
323{
324  void flush_if_large(uint64 per_kv, std::map< Tag_Index_Global_KVI, std::vector< Ref_Entry > >& db_to_insert)
325  {
326    total += per_kv;
327    if (total >= THRESHOLD_24)
328    {
329      callback->migration_flush();
330      for (auto& i : db_to_insert)
331        std::sort(i.second.begin(), i.second.end());
332      into_db.update({}, db_to_insert);
333      db_to_insert.clear();
334      total = 0;
335    }    
336  }
337  
338  void flush_single_kv(std::map< Tag_Index_Global_KVI, std::vector< Ref_Entry > >& db_to_insert)
339  {
340    callback->migration_flush_single_kv();
341    for (auto& i : db_to_insert)
342      std::sort(i.second.begin(), i.second.end());
343    into_db.update({}, db_to_insert);
344    db_to_insert.clear();
345  }
346  
347  void flush_the_rest(std::map< Tag_Index_Global_KVI, std::vector< Ref_Entry > >& db_to_insert)
348  {
349    callback->migration_flush();
350    for (auto& i : db_to_insert)
351      std::sort(i.second.begin(), i.second.end());
352    into_db.update({}, db_to_insert);
353  }
354  
355  Block_Backend< Target_Idx, Ref_Entry >& into_db;
356  uint64 total;
357  Osm_Backend_Callback* callback;
358};

361template< typename Ref_Entry, typename Src_Idx, typename Target_Idx >
362void migrate_loop(
363    Block_Backend< Src_Idx, Ref_Entry >& from_db, Block_Backend< Target_Idx, Ref_Entry >& into_db,
364    Osm_Backend_Callback* callback, std::map< String_Index, std::set< Frequent_Value_Entry > >& freq_to_insert)
365{
366  auto from_it = from_db.flat_begin();
367
368  std::map< Tag_Index_Global_KVI, std::vector< Ref_Entry > > db_to_insert;
369  auto to_it = db_to_insert.begin();
370
371  // We have two different flush criteria here:
372  // - total number of objects after a couple of relatively small kvs
373  // - number of objects alone for a sufficently large kv
374  // On top of this, we need to rebuild the index for the affected kv if a kv passes its size threshold.
375  Migrate_Loop_Flush< Ref_Entry, Target_Idx > flush{ into_db, 0, callback };
376  uint64 per_kv = 0;
377  uint64 level_threshold = THRESHOLD_8;
378  uint level = 0;
379  while (!(from_it == from_db.flat_end()))
380  {
381    if (to_it == db_to_insert.end() || to_it->first.key != from_it.index().key
382        || to_it->first.value != from_it.index().value)
383    {
384      if (level > 0)
385        freq_to_insert[{ to_it->first.key }].insert({ to_it->first.value, per_kv, level });
386
387      flush.flush_if_large(per_kv, db_to_insert);
388      
389      to_it = db_to_insert.insert({ Tag_Index_Global_KVI(from_it.index().key, from_it.index().value), {} }).first;
390      level_threshold = THRESHOLD_8;
391      level = 0;
392      per_kv = 0;
393    }
394    else if (per_kv >= level_threshold)
395    {
396      if (level < 24)
397      {
398        level = calc_tag_split_level(per_kv);
399        rebuild_db_to_insert(db_to_insert, from_it.index().key, from_it.index().value, level);
400        level_threshold = (level < 16 ? THRESHOLD_16 : THRESHOLD_24);
401      }
402      if (level == 24)
403      {
404        flush.flush_single_kv(db_to_insert);
405        
406        to_it = db_to_insert.insert({ Tag_Index_Global_KVI(from_it.index().key, from_it.index().value), {} }).first;
407        level_threshold += THRESHOLD_24;
408      }
409    }
410
411    if (level == 0)
412      to_it->second.push_back(from_it.object());
413    else
414      db_to_insert[Tag_Index_Global_KVI(
415          from_it.index().key, from_it.index().value, (from_it.object().idx.val() & (0xffffffff<<(32-level))))]
416          .push_back(from_it.object());
417    ++per_kv;
418    ++from_it;
419  }
420  if (to_it != db_to_insert.end() && level > 0)
421    freq_to_insert[{ to_it->first.key }].insert({ to_it->first.value, per_kv, level });
422
423  flush.flush_the_rest(db_to_insert);
424}
