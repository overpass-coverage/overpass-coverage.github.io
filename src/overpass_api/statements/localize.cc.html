605template< typename Index, typename Maybe_Attic >
606void generate_trigraphs(const std::map< Index, std::vector< Maybe_Attic > >& items, Set_With_Context& context_from,
607    Set& into, Resource_Manager& rman, NWR_Context& nwr_context,
608    Localize_Statement::Mode mode, const Owner< Bbox_Double >& bbox)
6090{
6100  for (typename std::map< Index, std::vector< Maybe_Attic > >::const_iterator it_idx = items.begin();
6110      it_idx != items.end(); ++it_idx)
6120  {
6130    for (typename std::vector< Maybe_Attic >::const_iterator it_elem = it_idx->second.begin();
6140        it_elem != it_idx->second.end(); ++it_elem)
6150    {
6160      const Element_With_Context< Maybe_Attic >& data = context_from.get_context(it_idx->first, *it_elem);
617
6180      if (mode == Localize_Statement::all || (data.tags && !data.tags->empty()))
6190      {
6200        Derived_Structure* last = 0;
6210        Point_Double last_point(100., 0.);
622
6230        for (std::vector< Relation_Entry >::const_iterator it_memb = data.object->members.begin();
6240            it_memb != data.object->members.end(); ++it_memb)
6250        {
6260          if (it_memb->type == Relation_Entry::NODE)
6270          {
6280            NWR_Context::Node_Context& node_context = nwr_context.context_by_node_id(it_memb->ref);
629
6300            if (admissible_by_bbox(bbox, node_context))
6310            {
6320              Derived_Structure_Builder result(rman, "trigraph",
6330                  new Point_Geometry(node_context.lat, node_context.lon), data.tags);
634
6350              if (node_context.local_id.val())
6360                result.set_tag("_local_ref", to_string(node_context.local_id.val()));
637
6380              if (mode == Localize_Statement::all)
6390              {
6400                result.set_tag("_child_node_ref", to_string(it_memb->ref.val()));
6410                result.set_tag("_relation_ref", to_string(it_elem->id.val()));
6420              }
643
6440              if (last)
6450              {
6460                last->tags.push_back(std::make_pair("_next", to_string(result.get().id.val())));
6470                result.set_tag("_previous", to_string(last->id.val()));
6480              }
6490              into.deriveds[Uint31_Index(0u)].push_back(result.get());
6500              last = &into.deriveds[Uint31_Index(0u)].back();
6510            }
6520            else
6530              last = 0;
6540          }
6550          else if (it_memb->type == Relation_Entry::WAY)
6560          {
6570            std::vector< NWR_Context::Way_Section_Context >& way_context
6580                = nwr_context.context_by_way_id(Way_Skeleton::Id_Type(it_memb->ref.val()));
659
6600            if (way_context.empty())
6610            {
6620              last = 0;
6630              continue;
6640            }
665
6660            enum { both_possible, must_forwards, must_backwards } elem_orientation = both_possible;
6670            if (last && last_point.lat <= 90.)
6680            {
6690              if (!way_context.front().points.empty() && last_point == way_context.front().points.front())
6700                elem_orientation = must_forwards;
6710              else if (!way_context.back().points.empty() && last_point == way_context.back().points.back())
6720                elem_orientation = must_backwards;
6730            }
6740            if (elem_orientation == both_possible)
6750            {
6760              std::vector< Relation_Entry >::const_iterator it_next = it_memb+1;
6770              if (it_next != data.object->members.end() && it_next->type == Relation_Entry::WAY)
6780              {
6790                std::vector< NWR_Context::Way_Section_Context >& next_way_context
6800                    = nwr_context.context_by_way_id(Way_Skeleton::Id_Type(it_next->ref.val()));
6810                if (!next_way_context.empty())
6820                {
6830                  if (!way_context.back().points.empty()
6840                    && ((!next_way_context.front().points.empty()
6850                        && next_way_context.front().points.front() == way_context.back().points.back())
6860                      || (!next_way_context.back().points.empty()
6870                        && next_way_context.back().points.back() == way_context.back().points.back())))
6880                    elem_orientation = must_forwards;
6890                  else if (!way_context.front().points.empty()
6900                    && ((!next_way_context.front().points.empty()
6910                        && next_way_context.front().points.front() == way_context.front().points.front())
6920                      || (!next_way_context.back().points.empty()
6930                        && next_way_context.back().points.back() == way_context.front().points.front())))
6940                    elem_orientation = must_backwards;
6950                }
6960              }
6970            }
698
6990            if (elem_orientation != must_backwards)
7000            {
7010              for (uint i = 0; i < way_context.size(); ++i)
7020              {
7030                if (way_context[i].points.empty())
7040                {
7050                  last = 0;
7060                  continue;
7070                }
708
7090                Derived_Structure_Builder result(rman, "trigraph",
7100                    new Linestring_Geometry(way_context[i].points), data.tags);
711
7120                if (way_context[i].local_from.val())
7130                  result.set_tag("_vertex_local_from", to_string(way_context[i].local_from.val()));
7140                if (way_context[i].local_to.val())
7150                  result.set_tag("_vertex_local_to", to_string(way_context[i].local_to.val()));
7160                if (way_context[i].local_id.val())
7170                  result.set_tag("_local_ref", to_string(way_context[i].local_id.val()));
718
7190                if (mode == Localize_Statement::all)
7200                {
7210                  result.set_tag("_child_way_ref", to_string(it_memb->ref.val()));
7220                  result.set_tag("_relation_ref", to_string(it_elem->id.val()));
7230                }
724
7250                if (last)
7260                {
7270                  last->tags.push_back(std::make_pair("_next", to_string(result.get().id.val())));
7280                  result.set_tag("_previous", to_string(last->id.val()));
7290                }
7300                into.deriveds[Uint31_Index(0u)].push_back(result.get());
7310                last = &into.deriveds[Uint31_Index(0u)].back();
7320              }
7330              last_point = (way_context.empty() || way_context.back().points.empty()) ? Point_Double(100., 0.)
7340                  : way_context.back().points.back();
7350            }
7360            else
7370            {
7380              for (int i = way_context.size()-1; i >= 0; --i)
7390              {
7400                if (way_context[i].points.empty())
7410                {
7420                  last = 0;
7430                  continue;
7440                }
745
7460                std::vector< Point_Double > points = way_context[i].points;
7470                std::reverse(points.begin(), points.end());
7480                Derived_Structure_Builder result(rman, "trigraph",
7490                    new Linestring_Geometry(points), data.tags);
750
7510                if (way_context[i].local_to.val())
7520                  result.set_tag("_vertex_local_from", to_string(way_context[i].local_to.val()));
7530                if (way_context[i].local_from.val())
7540                  result.set_tag("_vertex_local_to", to_string(way_context[i].local_from.val()));
7550                if (way_context[i].local_id.val())
7560                  result.set_tag("_local_ref", to_string(way_context[i].local_id.val()));
757
7580                if (mode == Localize_Statement::all)
7590                {
7600                  result.set_tag("_child_way_ref", to_string(it_memb->ref.val()));
7610                  result.set_tag("_relation_ref", to_string(it_elem->id.val()));
7620                }
763
7640                if (last)
7650                {
7660                  last->tags.push_back(std::make_pair("_next", to_string(result.get().id.val())));
7670                  result.set_tag("_previous", to_string(last->id.val()));
7680                }
7690                into.deriveds[Uint31_Index(0u)].push_back(result.get());
7700                last = &into.deriveds[Uint31_Index(0u)].back();
7710              }
7720              last_point = (way_context.empty() || way_context.front().points.empty()) ? Point_Double(100., 0.)
7730                  : way_context.front().points.front();
7740            }
7750          }
7760          else if (mode >= Localize_Statement::also_loose)
7770          {
7780            Derived_Structure_Builder result(rman, "trigraph", new Null_Geometry(), data.tags);
779
7800            if (mode == Localize_Statement::all)
7810            {
7820              result.set_tag("_child_relation_ref", to_string(it_memb->ref.val()));
7830              result.set_tag("_relation_ref", to_string(it_elem->id.val()));
7840            }
785
7860            if (last)
7870            {
7880              last->tags.push_back(std::make_pair("_next", to_string(result.get().id.val())));
7890              result.set_tag("_previous", to_string(last->id.val()));
7900            }
7910            into.deriveds[Uint31_Index(0u)].push_back(result.get());
7920            last = &into.deriveds[Uint31_Index(0u)].back();
7930          }
7940          else
795            // Relation members do not have any effect without loose mode
7960            last = 0;
7970        }
7980      }
7990    }
8000  }
8010}

8050{
806  //TODO: diff
807
8080  Set into;
809
8100  const Set* input_set = rman.get_set(input);
8110  if (!input_set)

8130    transfer_output(rman, into);
8140    return;
8150  }
816
8170  Owner< Bbox_Double > bbox(south <= 90. ? new Bbox_Double(south, west, north, east) : 0);
818

8200  requested_context.add_usage(input, Set_Usage::TAGS);

8270    process_vertices(context_from->base->nodes, *context_from, into, rman, nwr_context, type, bbox);
8280    if (!context_from->base->attic_nodes.empty())
8290      process_vertices(context_from->base->attic_nodes, *context_from, into, rman, nwr_context, type, bbox);
830
8310    nwr_context.prepare_links(rman, *this, *input_set, bbox);
8320    generate_links(context_from->base->ways, *context_from, into, rman, nwr_context, type, bbox);
8330    if (!context_from->base->attic_ways.empty())

835

8370    {

8410    }
842
8430    generate_trigraphs(context_from->base->relations, *context_from, into, rman, nwr_context, type, bbox);
8440    if (!context_from->base->attic_ways.empty())
8450      generate_trigraphs(context_from->base->attic_relations, *context_from, into, rman, nwr_context, type, bbox);
8460  }
